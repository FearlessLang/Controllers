main(desk, agent, test):
  rmTree(channel)
  build(portable, managed, commons, frontend, coordinator, controllerTest)
  junit(desk, agent, test or all)

walks():
  if (!agent && actions.any(a->!a.perDesk.has(desk))){ abort }
  if (agent){ cleanDir(channel) }
  last= now()
  walk()

action(name, ons...): a= Action(name, ons); actions.add(a); return a
on(desk, steps): (desk, steps)
n(recorded): agent ? values[index++] : recorded

Action.go():
  index= 0
  values= agent ? answer(name) : none
  perDesk[desk]()
  assert !agent || index==values.length

answer(name):
  file= channel/name
  write(file,"")
  loop:
    text= read(file).strip()
    if (text.endsWith(";")){ delete(file); return numbers(text) }
    for (request in channel/*.do){ serve(request) }
    pause(100)

serve(request):
  v= numbers(read(request))
  if (v.isEmpty()){ return }
  case request.name:
    "shot.do": write(channel/shot.png, pilot.shot())
    "hover.do": pilot.glide(v0,v1,none,v0,v1,none)
  delete(request)

until(done):
  end= now()+60000
  while (!done()){ assert now()<end; pause(100) }
  last= now()

ready(v):
  moment= agent ? v : last+v
  if (moment>now()){ pause(moment-now()) }
  last= now()

click(x,y): pilot.click(x,y)
doubleClick(x,y): pilot.doubleClick(x,y)
drag(x0,y0,x1,y1): pilot.drag(x0,y0,x1,y1)
keys(codes): pilot.chord(codes)

look():
  pilot.glide(right edge, middle, none, right edge, middle, none)
  return pilot.shot()

same(a,b): count(i: sum(abs(channelDiff(a[i],b[i])))>30)*20 < pixels(a)
differ(a,area,b): !same(crop(a,area),crop(b,area))

pilot.glide(x0,y0,held,x1,y1,then):
  mouseMove(x0,y0)
  hold(held)
  steps= max(abs(x1-x0),abs(y1-y0))/4+1
  for i in 1..steps: mouseMove(lerp(x0,x1,i/steps),lerp(y0,y1,i/steps)); pause(4)
  hold(then)

pilot.hold(want):
  for b in buttons: if (down(b)!=want(b)){ set(b,want(b)) }
  down= want
  pause(200)

pilot.click(x,y): glide(x,y,none,x,y,left); glide(x,y,left,x,y,none)

pilot.doubleClick(x,y):
  glide(x,y,none,x,y,none)
  twice: press(left); pause(50); release(left); pause(50)
  pause(200)

pilot.drag(x0,y0,x1,y1): glide(x0,y0,none,x0,y0,left); glide(x0,y0,left,x1,y1,left); glide(x1,y1,left,x1,y1,none)

pilot.chord(codes): press(codes); release(reverse(codes)); pause(200)

pilot.shot():
  img= capture(screen)
  assert !allBlack(img)
  return img

pilot.changed(a,b,r):
  differs= pixelwise(a,b,10)
  solid= differs where the whole (2r+1) square around it differs
  return box(largest connected component of solid, grown by r)
