final class ExampleTest extends PilotTest{
  final Action chooseFirst= action("chooseFirst",
    on("ubuntu_gnome",()->click(n(120),n(80))),
    on("windows",()->doubleClick(n(60),n(40))));
  final Action settled= action("settled",
    on("ubuntu_gnome",()->ready(n(2000))),
    on("windows",()->ready(n(500))));
  final Action chooseSecond= action("chooseSecond",
    on("ubuntu_gnome",()->click(n(300),n(200))),
    on("windows",()->click(n(150),n(90))));
  @Override protected void walk(){
    chooseFirst.go();
    settled.go();
    chooseSecond.go();
  }
}

record Action(String name, Map<String,Runnable> perDesk){
  void go(){ current= name; index= 0; perDesk.get(desk).run(); }
}

long n(long recorded){
  if (!agent){ return recorded; }
  return answer(current)[index++];
}

long[] answer(String name){
  file= channel/name
  create(file, empty)
  until(file ends with ";")
  return numbers(file)
}

click(x,y)       -> pilot.click(x,y)
doubleClick(x,y) -> pilot.doubleClick(x,y)
ready(ms)        -> sleep until last+ms
