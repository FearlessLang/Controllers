final class WindowCloseTest extends ManagerTest{
  final Action openManager= action("openManager",
    on("ubuntu_gnome",()->drag(n(helloWorldIcon.x),n(helloWorldIcon.y),n(launcherIcon.x),n(launcherIcon.y))),
    on("windows",()->drag(n(helloWorldIcon.x),n(helloWorldIcon.y),n(launcherIcon.x),n(launcherIcon.y))));
  final Action managerShown= action("managerShown",
    on("ubuntu_gnome",()->ready(n(2000))),
    on("windows",()->ready(n(2000))));
  final Action closeManager= action("closeManager",
    on("ubuntu_gnome",()->click(n(3822),n(50))),
    on("windows",()->click(n(1255),n(11))));
  final Action openLauncher= action("openLauncher",
    on("ubuntu_gnome",()->doubleClick(n(launcherIcon.x),n(launcherIcon.y))),
    on("windows",()->doubleClick(n(launcherIcon.x),n(launcherIcon.y))));
  final Action managerBack= action("managerBack",
    on("ubuntu_gnome",()->ready(n(1000))),
    on("windows",()->ready(n(1000))));
  final Action managerMenu= action("managerMenu",
    on("ubuntu_gnome",()->click(n(98),n(79))),
    on("windows",()->click(n(31),n(33))));
  final Action quitManager= action("quitManager",
    on("ubuntu_gnome",()->click(n(128),n(191))),
    on("windows",()->click(n(61),n(167))));
  final Action runningSign= action("runningSign",
    on("ubuntu_gnome",()->area(n(x),n(y),n(w),n(h))),
    on("windows",()->area(n(x),n(y),n(w),n(h))));
  @Override protected void walk(){
    var background= look();
    var sign= runningSign.area();
    openManager.go();
    managerShown.go();
    var shown= look();
    closeManager.go();
    until(()->same(background,look()));
    assertTrue(differ(background,sign,look()));
    openLauncher.go();
    managerBack.go();
    assertTrue(same(shown,look()));
    managerMenu.go();
    quitManager.go();
    until(()->!differ(background,sign,look()));
  }
}
