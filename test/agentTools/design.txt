  compile(controllerTest);

  separateProcess(()->junit(property: testName or all, desk, agent, channelFolder,filesIOFolder));

class ManagerTest{
  String desk= property("desk")
  boolean agent= property("agent")
  Path channel= property("channelFolder")
  Path filesIOFolder =property("filesIOFolder");
  abstract void walk() throws Throwable;
  Time start;
  int inputIndex=0;
  String inputName;
  List<Object> inputs;
  @Test void test() throws Throwable{
    start = now();
    walk();
  }
  Action action(String name, On...ons){ return new Action(this,name,ons.find(desk).orCrash); }
  On on(String desk, Run steps){ new On(desk,steps); }

}
record Action(ManagerTest m, String name, Run run){
  void go() throws Throwable{
    if(!m.agent){ run.run(); return; }
    m.name = name;
    m.inputIndex = 0;
    cleanDir(channelFolder);
    makeFile(channel,name);
    waitUntilFileEndsWithSemicolon(channel,name);
    inputs=makeInputs(channel,name);
    run.run(); 
  }
}
interface Run{ void run() throws Throwable; }
record On(String desk, Run steps){}

Object val(Object recorded){
  if(!agent){ return recorded;}
  return inputs.get(inputIndex++); 
}
List<Object> makeInputs(channel,name){
  String f= fileContent(channel,name);
  f.split(\n).stream()
    .map(String::trim)
    .map(s->s.startsWith("\"")?parseStr(s):parseInteger(s))
    .toList();
}
void waitUtilTime(Time time){
  Time current=now();
  while (start-current>time){}//implementation should not busy loop so fast
  return;
}

void checkContent(List<String> path,String expected){
  c= readContent(filesIOFolder+path);
  assert c is expected;
}
class ExampleTest extends ManagerTest{
  Action chooseFirst= action("chooseFirst",
    on("ubuntu_gnome",()->click(val(120),val(80))),
    on("windows",()->doubleClick(val(60),val(40))));
  Action settled= action("settled",
    on("ubuntu_gnome",()->waitUtilTime(val(2000))),
    on("windows",()->waitUtilTime(val(500))));
  Action chooseSecond= action("chooseSecond",
    on("ubuntu_gnome",()->click(val(300),val(200))),
    on("windows",()->click(val(150),val(90))));

  void walk(){
    writeTestSpecificFiles();
    stabilize();
    chooseFirst.go();
    settled.go();
    chooseSecond.go();
    stabilize();
    checkContent(List.of(..),"expected");
  }
