long[] answer(String name):
  Path file= channel/name
  write(file,"")
  loop:
    String text= read(file).strip()
    if (text.endsWith(";")){ delete(file); return numbers(text) }
    for (Path request in channel/*.do){ serve(request) }
    pause(100)

void serve(Path request):
  long[] v= numbers(read(request))
  if (v.isEmpty()){ return }
  case request.name:
    "shot.do": write(channel/shot.png, pilot.shot())
    "hover.do": pilot.glide(v0,v1,none,v0,v1,none)
  delete(request)

void until(BooleanSupplier done):
  long end= now()+60000
  while (!done()){ assert now()<end; pause(100) }
  last= now()

void ready(long v):
  long moment= agent ? v : last+v
  if (moment>now()){ pause(moment-now()) }
  last= now()

void pilot.glide(int x0, int y0, Set<Button> held, int x1, int y1, Set<Button> then):
  mouseMove(x0,y0)
  hold(held)
  int steps= max(abs(x1-x0),abs(y1-y0))/4+1
  for i in 1..steps: mouseMove(lerp(x0,x1,i/steps),lerp(y0,y1,i/steps)); pause(4)
  hold(then)

void pilot.hold(Set<Button> want):
  for b in buttons: if (down(b)!=want(b)){ set(b,want(b)) }
  down= want
  pause(200)

void pilot.click(int x, int y): glide(x,y,none,x,y,left); glide(x,y,left,x,y,none)

void pilot.doubleClick(int x, int y):
  glide(x,y,none,x,y,none)
  twice: press(left); pause(50); release(left); pause(50)
  pause(200)

void pilot.drag(int x0, int y0, int x1, int y1): glide(x0,y0,none,x0,y0,left); glide(x0,y0,left,x1,y1,left); glide(x1,y1,left,x1,y1,none)

void pilot.chord(int... codes): press(codes); release(reverse(codes)); pause(200)

BufferedImage pilot.shot():
  BufferedImage img= capture(screen)
  assert !allBlack(img)
  return img

Rectangle pilot.changed(BufferedImage a, BufferedImage b, int r):
  differs= pixelwise(a,b,10)
  solid= differs where the whole (2r+1) square around it differs
  return box(largest connected component of solid, grown by r)

class ExampleSeeTest:
  Action carry= action("carry",
    on("ubuntu_gnome",()->drag(n(300),n(500),n(3700),n(120))),
    on("windows",()->drag(n(200),n(400),n(900),n(60))))
  Action close= action("close",
    on("ubuntu_gnome",()->click(n(3822),n(50))),
    on("windows",()->click(n(1255),n(11))))
  void walk():
    BufferedImage before= look()
    carry.go()
    until(()->!same(before,look()))
    BufferedImage shown= look()
    close.go()
    until(()->same(before,look()))
    assert !same(shown,look())
